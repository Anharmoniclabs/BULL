cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD adafruit_kb2040 CACHE STRING "Target board")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(bull_hardware_authority C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(bull_hardware_diagnostic src/main.c src/usb_descriptors.c src/state_machine.c src/boot_clicks.c src/boot_button.c)
target_include_directories(bull_hardware_diagnostic PRIVATE src)
target_link_libraries(bull_hardware_diagnostic pico_stdlib hardware_i2c hardware_pio tinyusb_device tinyusb_board)
pico_generate_pio_header(bull_hardware_diagnostic ${CMAKE_CURRENT_LIST_DIR}/src/status_pixel.pio)
pico_enable_stdio_usb(bull_hardware_diagnostic 0)
pico_enable_stdio_uart(bull_hardware_diagnostic 0)
pico_add_extra_outputs(bull_hardware_diagnostic)
